Q1 = df.select(
    (pl.col("num_of_delays_carrier").sum()).alias("Carrier"),
    (pl.col("num_of_delays_security").sum()).alias("Security"),
    (pl.col("num_of_delays_late_aircraft").sum()).alias("Late Aircraft")
).with_columns(
    (pl.col("Carrier") + pl.col("Security") + pl.col("Late Aircraft")).alias("Grand_total")
).with_columns(
    (pl.col("Carrier") / pl.col("Grand_total")).alias("Carrier_Proportion"),
    (pl.col("Security") / pl.col("Grand_total")).alias("Security_Proportion"),
    (pl.col("Late Aircraft") / pl.col("Grand_total")).alias("Late_Aircraft_Proportion")
).drop("Carrier", "Security", "Late Aircraft", "Grand_total")

display(Q1)

# Now I'll unpivot it so I can make a nice bar chart
Q1_long = Q1.unpivot(
    on=["Late_Aircraft_Proportion", "Carrier_Proportion", "Security_Proportion"], # I put them in order here 
    variable_name="Delay_Type",
    value_name="Proportion"
)

# display(Q1_long)

(
    ggplot(data=Q1_long, mapping=aes(x="Delay_Type", y="Proportion"))
    + geom_bar(stat="identity")

     (I'm deffinitely doing stacked bar charts, looks cool)

- group by airport_code for airport isolation
- sum the 3 delay columns
- get grand total for each airport
- compute proportions per airport
- unpivot so each row is a delay type
- plot stacked bar chart (x=airport, x=proportion)
"""

display(df)

Q2_wide = df.group_by("airport_code").agg(
    pl.col("num_of_delays_carrier").sum(),
    pl.col("num_of_delays_security").sum(),
    pl.col("num_of_delays_late_aircraft").sum()
).with_columns(
    pl.sum_horizontal(
        "num_of_delays_carrier", "num_of_delays_security", "num_of_delays_late_aircraft"
    ).alias("Grand_total")
).with_columns(
    (pl.col("num_of_delays_carrier") / pl.col("Grand_total")).alias("Carrier_Proportion"),
    (pl.col("num_of_delays_security") / pl.col("Grand_total")).alias("Security_Proportion"),
    (pl.col("num_of_delays_late_aircraft") / pl.col("Grand_total")).alias("Late_Aircraft_Proportion"),
).drop(
    "num_of_delays_carrier", "num_of_delays_security", "num_of_delays_late_aircraft", "Grand_total"
)

# display(Q2_wide)

Q2_long = Q2_wide.unpivot(
    index=["airport_code"],
    on=["Carrier_Proportion", "Security_Proportion", "Late_Aircraft_Proportion"],
    variable_name="Delay_Type",
    value_name="Proportion"
)

# display(Q2_long)

(
    ggplot(data=Q2_long, mapping=aes(x="airport_code", y="Proportion", fill="Delay_Type"))
    + geom_bar(stat="identity")

    + theme_gray()
